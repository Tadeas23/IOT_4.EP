from machine import Pin, UART
import time

# UART nastavení (UART0 na pinech GP0=TX, GP1=RX)
uart = UART(0, baudrate=9600, tx=Pin(0), rx=Pin(1))

# Piny
button = Pin(14, Pin.IN, Pin.PULL_DOWN)
led = Pin(15, Pin.OUT)

last_state = 0

while True:
    # --- Odesílání ---
    state = button.value()
    if state != last_state:  # změna stavu tlačítka
        if state == 1:
            uart.write("ON\n")
        else:
            uart.write("OFF\n")
        last_state = state

    # --- Příjem ---
    if uart.any():
        msg = uart.readline().decode().strip()
        if msg == "ON":
            led.value(1)
        elif msg == "OFF":
            led.value(0)

    time.sleep(0.05)
