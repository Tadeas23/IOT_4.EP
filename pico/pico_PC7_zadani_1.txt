########################### komunikace přes repl přes usb, zkouška jak to vůbec rozjet na thonnym a poslat to do pica, tak aby to fungovalo
from machine import Pin
import sys

led = Pin(2, Pin.OUT)

print("Pico ready, type 'red' to turn on LED, 'off' to turn off.")

while True:
    cmd = input(">> ")  # čeká na zadání příkazu v REPL
    if cmd.lower() == "red":
        led.value(1)
        print("LED ON")
    elif cmd.lower() == "off":
        led.value(0)
        print("LED OFF")
    else:
        print("Unknown command")
